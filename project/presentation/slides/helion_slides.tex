\begin{frame}[t]{What the evidence supports next}
  \lead{No incremental ML diagnostic value demonstrated\newline
    over candidate rules under these assumptions.}
  \vspace{1mm}
  \begin{description}
    \setlength{\itemsep}{2.4mm}
    \item[Prediction] Investigate weak-fault errors, available features and
      regularisation using training-lot validation.
    \item[Decision] Qualify the real SOP, useful alternative paths,
      examination scope and continuation costs.
    \item[Evidence] Replace operational assumptions and collect representative
      complete audits plus fresh prospective evidence.
  \end{description}
  \vspace{3mm}
  \smalltext{\textbf{Retaining rules remains valid.} Any shadow study or controlled pilot
    needs qualification. No policy is selected for operational release.}
  \source{Benchmark limitations and next evidence; client brief audit requirement. Improvement experiments are proposed.}
